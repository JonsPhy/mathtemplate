% ---------------------------------------------------------------------------
%  Box geometry
% ---------------------------------------------------------------------------
%  projectboxbase is shared by every numbered box. style/slides.sty appends to
%  it (no page breaks, smaller title) so the same boxes fit on a slide.
% ---------------------------------------------------------------------------
\tcbset{
    projectboxbase/.style={
        enhanced,
        breakable,
        projectboxframe,
        titlerule=0mm,
        toptitle=1mm,
        bottomtitle=1mm,
        fonttitle=\bfseries\large,
        coltitle=black,
    },
    projectdefnbox/.style={projectboxbase, colbacktitle=projectdefntitle, colback=projectdefnback},
    projectthmbox/.style={projectboxbase,  colbacktitle=projectthmtitle,  colback=projectthmback},
    projectlembox/.style={projectboxbase,  colbacktitle=projectlemtitle,  colback=projectlemback},
    projectcorbox/.style={projectboxbase,  colbacktitle=projectcortitle,  colback=projectcorback},
    projectpropbox/.style={projectboxbase, colbacktitle=projectproptitle, colback=projectpropback},
    projectclmbox/.style={projectboxbase,  colbacktitle=projectclmtitle,  colback=projectclmback},
    projectfactbox/.style={projectboxbase, colbacktitle=projectfacttitle, colback=projectfactback},
    projectexcbox/.style={projectboxbase,  colbacktitle=projectexctitle,  colback=projectexcback},
    projecthwbox/.style={projectboxbase,   colbacktitle=projecthwtitle,   colback=projecthwback},
}
