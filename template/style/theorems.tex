% ===========================================================================
%  Theorem system
% ---------------------------------------------------------------------------
%  One command set, two renderings. The look is chosen by a package option and
%  nothing in your content ever changes:
%
%    colour / mono   coloured (or grey) tcolorbox boxes  -> theorems-boxed.tex
%    paper           plain amsthm environments, no boxes -> theorems-paper.tex
%
%  Numbered (all share one counter, numbered within the section):
%    Definition   \defn{Title}{body}          Theorem      \thm{Title}{body}
%    Lemma        \lem{Title}{body}           Corollary    \cor{body}
%    Proposition  \prop{body}                 Fact         \fact{body}
%    Claim        \clm{Title}{body}           (unnumbered)
%
%  Unnumbered:
%    Example      \ex{body}                   Remark        \rmkb{body}
%    Fact         \factb{body}                Inline remark \rmk{text}
%    Proof        \pf{body}
%
%  Suffixes:
%    r   referenceable: adds a label argument, e.g.
%          \defnr{Title}{label}{body}  ->  \ref{defn:label}
%          \thmr{Title}{label}{body}   ->  \ref{thm:label}
%    p   attaches a proof block, e.g.
%          \lemp{Title}{body}{proof}, \corp{body}{proof}, \propp{body}{proof},
%          \thmp{Title}{body}{proof}, \clmp{Title}{body}{proof}
%    pr  both, e.g. \thmpr{Title}{label}{body}{proof}
%
%  An empty title argument is allowed: \lem{}{...} renders just "Lemma".
%
%  \defn indexes its title automatically. Override the key with
%  \defn[key]{Title}{body}; suppress the entry with \defn[-]{Title}{body}.
%
%  All body arguments are declared "+m", so display math, align, itemize,
%  tikzpicture, ... work inside them without extra bracing. The flip side is
%  that \verb does not work there - write \texttt{\textbackslash cmd}.
%
%  Note: every numbered environment shares one counter, so in the boxed look
%  cleveref cannot infer the type name --- use plain \ref{thm:label} and write
%  the word ("Theorem \ref{thm:maschke}") yourself. That works in both looks.
% ===========================================================================
